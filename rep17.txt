
        // Fetch Expenses

        for (var e in exps) {
          totalExpenses += e.amount;
        }

        double finalNetProfit = netSales - totalExpenses;

          startDate: start,

          totalGrossSales: grossSales,

          totalNetSales: netSales,
          totalExpenses: totalExpenses,
          finalNetProfit: finalNetProfit,

          totalKasbon: totalKasbon,
