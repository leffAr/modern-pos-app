        double grossSales = 0;
        double discounts = 0;
        double netSales = 0;
        double totalCash = 0;
        double totalQris = 0;
        double totalKasbon = 0;

        for (var p in payments) {
          final payment = p.readTable(appDb.payments);
          if (payment.method == "CASH") {
             totalCash += (payment.amount - payment.changeAmount);
          } else if (payment.method == "QRIS") {
             totalQris += payment.amount;
          } else if (payment.method == "KASBON") {
             totalKasbon += payment.amount;
          }
        }
        
        for (var t in txs) {
          // Find if this transaction is KASBON
          final isKasbon = payments.any((p) => p.readTable(appDb.payments).transactionId == t.id && p.readTable(appDb.payments).method == "KASBON");
          
          if (!isKasbon) {
            grossSales += t.subtotal;
            discounts += t.discount;
            netSales += t.grandTotal;
          }
        }
